008    package com.nokia.midp.example.jsr234.audio3d;
009    
010    /**
011     * Bean Class stored Distance attenuation attributes. Main attributes of Distance Attenuation are: <br>
012     * <ol>
013     * 1. Minimum distance<br>
014     * 2. Maximum distance<br>
015     * 3. Roll Off Factor<br>
016     * 4. Mute sound after max distance condition
017     * </ol>
018     * @author krn
019     */
020    
021    public class DistAttenuationAttributes {
022    
023            private int daRMin;
024            private int daRMax;
025            private int rollOfFactor;
026            private boolean daMuteAfterMax;
027    
028            //DistAttenuationAttributes is instantiated with default values
029            public DistAttenuationAttributes(int rmin, int rmax, int rf, boolean MaM) {
030                    daRMin = rmin;
031                    daRMax = rmax;
032                    rollOfFactor = rf;
033                    daMuteAfterMax = MaM;
034    
035            }
036    
037            public int getDaRMin() {
038                    return daRMin;
039            }
040    
041            public void setDaRMin(int daRMin) {
042                    this.daRMin = daRMin;
043            }
044    
045            public int getDaRMax() {
046                    return daRMax;
047            }
048    
049            public void setDaRMax(int daRMax) {
050                    this.daRMax = daRMax;
051            }
052    
053            public int getRollOfFactor() {
054                    return rollOfFactor;
055            }
056    
057            public void setRollOfFactor(int rollOfFactor) {
058                    this.rollOfFactor = rollOfFactor;
059            }
060    
061            public boolean isDaMuteAfterMax() {
062                    return daMuteAfterMax;
063            }
064    
065            public void setDaMuteAfterMax(boolean daMuteAfterMax) {
066                    this.daMuteAfterMax = daMuteAfterMax;
067            }
068    
069    }
