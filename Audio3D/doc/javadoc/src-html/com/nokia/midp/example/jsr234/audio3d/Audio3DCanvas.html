001    package com.nokia.midp.example.jsr234.audio3d;
002    
003    import javax.microedition.lcdui.Canvas;
004    import javax.microedition.lcdui.Graphics;
005    /**
006     * Canvas class to display User Interface with the sound Source movement. 
007     * UI gives facility to move sound source 3D to see the sound effect using UP,DOWN.LEFT and RIGHT keys.
008     * 
009     * @author gothadiy
010     *
011     */
012    public class Audio3DCanvas extends Canvas {
013    
014            public static int MAX_X;
015            public static int MAX_Y;
016    
017            /**Instance of Coordinates class*/
018            Coordinates coordinates;
019    
020            /**Audio3DMIDlet instance for display reference*/
021            Audio3DMIDlet midlet;
022    
023            /**
024             * Constructor to initialize all required coordinates
025             * @param maxValues for movement
026             * @param midlet Audio3DMIDlet instance
027             */
028            public Audio3DCanvas(int maxValues, Audio3DMIDlet midlet) {
029                    setTitle("AMMS Audio 3D");
030                    MAX_X = maxValues - 1;
031                    MAX_Y = (maxValues * getHeight()) / getWidth() - 1;
032    
033                    coordinates = new Coordinates(MAX_X, MAX_Y);
034    
035                    this.midlet = midlet;
036            }
037    
038            /**
039             * Method changes the location of sound source based on the key press.
040             * It also change the effect of LocationControl and DistanceAttenuationControl.
041             * @param key keyCode from the keyPress method
042             * @param controller Audio3DMIDlet instance
043             */
044            public void changeSourceLocation(int key, Audio3DMIDlet controller) {
045    
046                    // Handle Sound source navigation
047                    switch (key) {
048                    case UP:
049                            coordinates.sourceY--;
050                            if (coordinates.sourceY < 0) {
051                                    coordinates.sourceY = 0;
052                            }
053                            break;
054    
055                    case LEFT:
056                            coordinates.sourceX--;
057                            if (coordinates.sourceX < 0) {
058                                    coordinates.sourceX = 0;
059                            }
060                            break;
061    
062                    case DOWN:
063                            coordinates.sourceY++;
064                            if (coordinates.sourceY >= MAX_Y) {
065                                    coordinates.sourceY = MAX_Y;
066                            }
067                            break;
068    
069                    case RIGHT:
070                            coordinates.sourceX++;
071                            if (coordinates.sourceX >= MAX_X) {
072                                    coordinates.sourceX = MAX_X;
073                            }
074                            break;
075                    }
076    
077                    repaint();
078    
079                    // Set Control values
080                    setCartesians(controller);
081    
082            }
083            
084            /**
085             * Method sets LocationControl and DistanceAttenuationControl
086             * @param controller Audio3DMIDlet instance
087             */
088            private void setCartesians(Audio3DMIDlet controller) {
089                    controller.sourceLC.setCartesian(this.coordinates.sourceX
090                                    * controller.locAttr.getLocScale(), controller.locAttr
091                                    .getLocAltitude(), this.coordinates.sourceY
092                                    * controller.locAttr.getLocScale());
093    
094                    controller.specLC.setCartesian(this.coordinates.specX
095                                    * controller.locAttr.getLocScale(), 0, this.coordinates.specY
096                                    * controller.locAttr.getLocScale());
097                    
098                    controller.ctlDistAtt.setParameters(midlet.ctlDistAtt.getMinDistance(),
099                    midlet.ctlDistAtt.getMaxDistance(),
100                    midlet.ctlDistAtt.getMuteAfterMax(),
101                    midlet.ctlDistAtt.getRolloffFactor());
102                    
103            }
104    
105            /**
106             * Displays Audio 3D UI. It contains:<br>
107             * - Draw back ground<br>
108             * - Draw Sound source<br>
109             * - Draw Spectator<br>
110             */
111            protected void paint(Graphics graphics) {
112                    // Draw back ground
113                    graphics.setColor(50, 50, 100);
114                    graphics.fillRect(graphics.getClipX(), graphics.getClipY(), graphics
115                                    .getClipWidth(), graphics.getClipHeight());
116    
117                    // Draw Sound source
118                    graphics.setColor(0, 200, 0);
119                    graphics.fillRect((getWidth() * coordinates.sourceX) / MAX_X,
120                                    (getHeight() * coordinates.sourceY) / MAX_Y,
121                                    getWidth() / MAX_X, getHeight() / MAX_Y);
122    
123                    // Draw Spectator
124                    graphics.setColor(200, 0, 0);
125                    graphics.fillRect((getWidth() * coordinates.specX) / MAX_X,
126                                    (getHeight() * coordinates.specY) / MAX_Y, getWidth() / MAX_X,
127                                    getHeight() / MAX_Y);
128    
129            }
130    
131            /**
132             * identify key pressed and handle it
133             */
134            public void keyPressed(int keyCode) {
135    
136                    int key = getGameAction(keyCode);
137                    changeSourceLocation(key, midlet);
138    
139            }
140    
141    }
